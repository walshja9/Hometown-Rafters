> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eight generated clubs play a full 154-game season with standings, season stats and leaders, basic transactions through one valuation formula, an offseason that develops players and rolls to the next year, and the Reds' games playable through the M1 engine. Plus the three M1 review fixes.

**Architecture:** A pure-TypeScript `src/league/` layer on top of `src/baseball/`. `LeagueState` is plain JSON (clubs, rosters, schedule, results, season stats, pool, the live game, history, RNG streams) and every mutation is a function `(L, ...) => void` on a cloned state, the same discipline as `applyAction`. Games are simulated with `simulateGame` seeded from `${seed}:${year}:${day}:${home}`, so the league RNG streams are never touched during a season and viewing never draws. Saves become version 2 (league), and a save failure is shown, not swallowed.

**Tech Stack:** TypeScript (strict), Vite, React 19, Vitest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-19-era1-playable-slice-design.md` §6 (league and players), §7 (ledger only), §8 (valuation spread), §9 (outputs), §11 (saves), §13 (M2).

export function deserialize(json: string): GameState {
  const s = JSON.parse(json) as Partial<GameState> | null;
  if (!s || s.version !== SAVE_VERSION) throw new Error(`unsupported save version ${String(s?.version)}`);
  if (!s.teams || !s.sides || !s.rng || !Array.isArray(s.events)) throw new Error('save is missing game state');
  return s as GameState;
}

export function writeSave(kv: KV, s: GameState): void {
  const current = kv.getItem(KEY);
  if (current !== null) {
    try { deserialize(current); kv.setItem(PREV, current); } catch { /* never rotate a bad save over a good one */ }
  }
  kv.setItem(KEY, serialize(s));
}

```ts
import type { Batting, Pitching } from '../baseball/types';
import type { LeaguePlayer, Personality } from './types';

export const batValue = (b: Batting) => 0.3 * b.contact + 0.25 * b.power + 0.2 * b.eye + 0.1 * b.avoidK + 0.15 * b.speed;
export const pitchValue = (p: Pitching) => (p.control + p.stuff + p.contactSuppression) / 3;

// One number per player, on the rating scale. Premium defenders get a little; stamina matters a little.
export function overall(p: LeaguePlayer): number {
  if (p.pitching) return pitchValue(p.pitching) + 0.1 * (p.pitching.stamina - 65);
  return batValue(p.batting) + 0.1 * (p.batting.fielding - 65) + (p.pos === 'C' || p.pos === 'SS' ? 2 : 0);
}

export const BASE_VALUE = 6000; // dollars for a 65 player in his prime (~1920 average salary); M3's money research rescales by year
export const SPREAD = 0.15;     // every deal crosses this spread, so buy-low/sell-high can't print money
const ageFactor = (age: number) => (age <= 23 ? 1.1 : age <= 29 ? 1 : age <= 32 ? 0.8 : age <= 35 ? 0.55 : 0.35);

export function value(p: LeaguePlayer): number {
  const now = overall(p);
  const eff = p.age < 27 ? 0.6 * now + 0.4 * p.potential : now;
  return Math.max(500, Math.round((BASE_VALUE * Math.exp(0.1 * (eff - 65)) * ageFactor(p.age)) / 100) * 100);
}
export const askPrice = (v: number) => Math.round(v * (1 + SPREAD));
export const bidPrice = (v: number) => Math.round(v * (1 - SPREAD));
// How much more than the spread each owner wants to see before saying yes.
export const GREED: Record<Personality, number> = { cheapskate: 1.6, gambler: 0.6, loyalist: 1, builder: 1 };
```

export function gauss(r: RngState): number {
  const u = 1 - next(r), v = next(r);
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
}
const clamp = (x: number, lo = 20, hi = 99) => Math.min(hi, Math.max(lo, Math.round(x)));
export const rating = (r: RngState, mean: number, sd = 9) => clamp(mean + sd * gauss(r));
export const between = (r: RngState, lo: number, hi: number) => lo + Math.floor(next(r) * (hi - lo + 1));

const POSITIONS: Position[] = ['C', '1B', '2B', '3B', 'SS', 'LF', 'CF', 'RF'];
const BENCH: Position[] = ['C', 'SS', '2B', 'CF', 'LF'];

export function genPlayer(L: LeagueState, pos: Position, mean: number, age: number, staminaMean = 65): LeaguePlayer {
  const r = L.rng.league;
  const id = `p${++L.seq}`;
  const name = `${FIRST[between(r, 0, FIRST.length - 1)]} ${LAST[between(r, 0, LAST.length - 1)]}`;
  const bats = next(r) < 0.3 ? 'L' : 'R';
  const throws = pos === 'P' ? (next(r) < 0.28 ? 'L' : 'R') : bats === 'L' && next(r) < 0.6 ? 'L' : 'R';
  const batting: Batting = pos === 'P'
    ? { ...PITCHER_BATTING, contact: rating(r, PITCHER_BATTING.contact, 5), power: rating(r, PITCHER_BATTING.power, 5) }
    : { contact: rating(r, mean), power: rating(r, mean), eye: rating(r, mean), avoidK: rating(r, mean), speed: rating(r, 65, 10), fielding: rating(r, 65, 10), arm: rating(r, 65, 10) };
  const p: LeaguePlayer = { id, name, bats, throws, batting, pos, age, potential: 0 };
  if (pos === 'P') {
    const pitching: Pitching = { control: rating(r, mean), stuff: rating(r, mean), contactSuppression: rating(r, mean), stamina: rating(r, staminaMean, 7) };
    p.pitching = pitching;
  }
  const now = overall(p);
  p.potential = age < 27 ? clamp(now + Math.max(0, (27 - age) * 1.5 + 4 * gauss(r)), Math.ceil(now), 99) : Math.ceil(now);
  return p;
}

// 25 men: 8 starters at the tier's mean, 5 bench bats 10 below, 4 starting pitchers at the mean with
// starter stamina, 8 relievers 7 below. Tier 2 is the league average; tiers 1 and 3 sit 3 points either side.
export function genRoster(L: LeagueState, tier: Tier): LeaguePlayer[] {
  const r = L.rng.league;
  const mean = 65 + 3 * (tier - 2);
  const roster: LeaguePlayer[] = [];
  for (const pos of POSITIONS) roster.push(genPlayer(L, pos, mean, between(r, 22, 33)));
  for (const pos of BENCH) roster.push(genPlayer(L, pos, mean - 10, between(r, 20, 34)));
  for (let i = 0; i < 12; i++) roster.push(genPlayer(L, 'P', i < 4 ? mean : mean - 7, between(r, 21, 34), i < 4 ? 78 : 55));
  return roster;
}

// The independent clubs' shop window: spare parts and a few kids, priced above value.

  return Array.from({ length: n }, (_, i) => {
    const p = genPlayer(L, cycle[i % cycle.length]!, 52, between(r, 17, 30), 62);
    return { ...p, ask: askPrice(value(p)) };
  });
}

import type { LineupSlot, Position, Team } from '../baseball/types';
import type { Club, LeaguePlayer } from './types';
import { batValue, pitchValue } from './valuation';

export const hitters = (roster: LeaguePlayer[]) => roster.filter((p) => !p.pitching);
export const pitchers = (roster: LeaguePlayer[]) => roster.filter((p) => p.pitching);

const FIELD: Position[] = ['C', 'SS', '2B', 'CF', '1B', '3B', 'LF', 'RF']; // premium positions are filled first
const PREMIUM = new Set<Position>(['C', 'SS', '2B', 'CF']);
const ORDER = [2, 3, 0, 1, 4, 5, 6, 7]; // batting slot for the i-th best bat: the best two hit third and fourth

// Best available man at each position (his own position first), big bats third and fourth, pitcher ninth.
export function autoLineup(club: Club): void {
  const taken = new Set<string>();
  const picks: { pos: Position; p: LeaguePlayer }[] = [];
  const fit = (p: LeaguePlayer, pos: Position) =>
    batValue(p.batting) + (PREMIUM.has(pos) ? 0.3 : 0.1) * p.batting.fielding - (p.pos === pos ? 0 : 8);
  for (const pos of FIELD) {
    const pool = hitters(club.roster).filter((p) => !taken.has(p.id));
    if (!pool.length) throw new Error(`${club.id} has no one left to play ${pos}`);
    const best = pool.reduce((a, b) => (fit(b, pos) > fit(a, pos) ? b : a));
    taken.add(best.id);
    picks.push({ pos, p: best });
  }
  picks.sort((a, b) => batValue(b.p.batting) - batValue(a.p.batting));
  const slots: LineupSlot[] = Array.from({ length: 9 }, () => ({ playerId: '', pos: 'P' as Position }));
  picks.forEach(({ pos, p }, i) => { slots[ORDER[i]!] = { playerId: p.id, pos }; });
  slots[8] = { playerId: club.rotation[0] ?? '', pos: 'P' };
  club.lineup = slots;
}

const spValue = (p: LeaguePlayer) => pitchValue(p.pitching!) + 0.3 * p.pitching!.stamina;
export function autoRotation(club: Club): void {
  club.rotation = pitchers(club.roster).sort((a, b) => spValue(b) - spValue(a)).slice(0, 4).map((p) => p.id);
  club.nextStarter = 0;
}

// Null when the lineup can take the field; otherwise what is wrong, in plain words for the roster screen.
export function lineupProblem(club: Club): string | null {
  const l = club.lineup;
  if (l.length !== 9) return 'a lineup needs nine men';
  const ids = new Set<string>(), positions = new Set<Position>();
  for (const s of l) {
    const p = club.roster.find((x) => x.id === s.playerId);
    if (!p) return `${s.playerId} is not on the roster`;
    if (ids.has(s.playerId)) return `${p.name} is listed twice`;
    ids.add(s.playerId);
    if (s.pos !== 'P' && p.pitching) return `${p.name} is a pitcher, not a ${s.pos}`;
    if (s.pos === 'P' && !p.pitching) return `${p.name} is not a pitcher`;
    if (positions.has(s.pos)) return `two men at ${s.pos}; every position needs one`;
    positions.add(s.pos);
  }
  return positions.size === 9 ? null : 'a position is missing';
}

// The engine's Team for today: the rotation's next man takes the pitcher's slot.
export function toTeam(club: Club, name: string): Team {
  if (!club.rotation.length) throw new Error(`${club.id} has no rotation`);
  const sp = club.rotation[club.nextStarter % club.rotation.length]!;
  const lineup = club.lineup.map((s) => (s.pos === 'P' ? { playerId: sp, pos: 'P' as Position } : s));
  return { id: club.id, name, players: club.roster, lineup, startingPitcherId: sp };
}

import type { Game } from './types';

export const DAYS = 154;
const DAY_MS = 86_400_000;

// Game day d falls on April 16 + d + one rest day every six games: 154 games over ~179 days, ending in October.
export function dateOf(year: number, day: number): string {
  return new Date(Date.UTC(year, 3, 16) + (day + Math.floor(day / 6)) * DAY_MS).toISOString().slice(0, 10);
}
export const isSunday = (date: string) => new Date(`${date}T00:00:00Z`).getUTCDay() === 0;

// Circle-method round robin: one club fixed, the other seven rotate, seven rounds a cycle, 22 cycles.
// A pair meets once per cycle in the same round, and the park alternates by cycle: 22 games, 11 each.
export function makeSchedule(ids: string[], year: number): Game[] {
  if (ids.length !== 8) throw new Error('the league is eight clubs');
  const games: Game[] = [];
  let ring = ids.slice(1);
  for (let round = 0; round < 7; round++) {
    const list = [ids[0]!, ...ring];
    for (let cycle = 0; cycle < 22; cycle++) {
      const day = cycle * 7 + round;
      for (let i = 0; i < 4; i++) {
        const a = list[i]!, b = list[7 - i]!;
        const flip = (cycle + i) % 2 === 1;
        games.push({ day, date: dateOf(year, day), home: flip ? b : a, away: flip ? a : b });
      }
    }
    ring = [ring[ring.length - 1]!, ...ring.slice(0, -1)];
  }
  return games.sort((x, y) => x.day - y.day);
}

import type { BatLine, BoxScore, PitchLine } from '../baseball/boxScore';
import type { PlayerSeason } from './types';

const zeroBat = (id: string): BatLine => ({ playerId: id, pa: 0, ab: 0, r: 0, h: 0, d: 0, t: 0, hr: 0, rbi: 0, bb: 0, k: 0, hbp: 0, sh: 0, sf: 0, sb: 0, cs: 0 });
const zeroPit = (id: string): PitchLine => ({ playerId: id, outs: 0, bf: 0, h: 0, r: 0, er: 0, bb: 0, k: 0, hr: 0, pitches: 0 });

export function season(stats: Record<string, PlayerSeason>, id: string): PlayerSeason {
  return (stats[id] ??= { g: 0, gp: 0, bat: zeroBat(id), pit: zeroPit(id) });
}

function add(into: BatLine | PitchLine, from: BatLine | PitchLine): void {
  const a = into as unknown as Record<string, number>, b = from as unknown as Record<string, number>;
  for (const k of Object.keys(b)) if (k !== 'playerId') a[k] = (a[k] ?? 0) + b[k]!;
}

// Fold one game's box score into season totals. Only totals are kept; the per-game box is not stored.
export function addBox(stats: Record<string, PlayerSeason>, box: BoxScore): void {
  for (const side of ['away', 'home'] as const) {
    for (const l of box.batting[side]) { const s = season(stats, l.playerId); s.g++; add(s.bat, l); }
    for (const l of box.pitching[side]) { const s = season(stats, l.playerId); s.gp++; add(s.pit, l); }
  }
}

const rate = (n: number, d: number) => (d ? (n / d).toFixed(3).replace(/^0/, '') : '.000');
export const avg = (b: BatLine) => rate(b.h, b.ab);
export const obp = (b: BatLine) => rate(b.h + b.bb + b.hbp, b.ab + b.bb + b.hbp + b.sf);
export const ip = (p: PitchLine) => `${Math.floor(p.outs / 3)}.${p.outs % 3}`;
export const era = (p: PitchLine) => (p.outs ? ((9 * p.er) / (p.outs / 3)).toFixed(2) : '0.00');

import type { GameState } from '../baseball/types';
import { newGame } from '../baseball/game';
import { simulateGame } from '../baseball/sim';
import { boxScore } from '../baseball/boxScore';
import { clubName } from './cities';
import { autoLineup, autoRotation, lineupProblem, toTeam } from './lineup';
import { DAYS } from './schedule';
import { addBox } from './stats';
import type { Club, LeagueState, PlayerSeason } from './types';

export const gameSeed = (L: LeagueState, home: string) => `${L.seed}:${L.year}:${L.day}:${home}`;

// A club whose rotation or lineup no longer fits its roster (a trade, a retirement) gets a fresh one.
function ready(club: Club): void {
  if (!club.rotation.length || club.rotation.some((id) => !club.roster.some((p) => p.id === id))) autoRotation(club);
  if (lineupProblem(club)) autoLineup(club);
}

function record(L: LeagueState, s: GameState): void {
  L.results.push({ day: L.day, home: s.teams.home.id, away: s.teams.away.id, homeRuns: s.score.home, awayRuns: s.score.away });
  addBox(L.stats, boxScore(s));
  for (const id of [s.teams.home.id, s.teams.away.id]) {
    const c = L.clubs[id]!;
    c.nextStarter = (c.nextStarter + 1) % c.rotation.length;
  }
}

// Plays the day. The user's game is opened as L.live (unless simUser) and the day closes in closeLive().
export function playDay(L: LeagueState, simUser = false): void {
  if (L.live) throw new Error('a game is in progress');
  if (L.day >= DAYS) throw new Error('the season is over');
  for (const club of Object.values(L.clubs)) ready(club);
  for (const g of L.schedule.filter((x) => x.day === L.day)) {
    const away = toTeam(L.clubs[g.away]!, clubName(g.away)), home = toTeam(L.clubs[g.home]!, clubName(g.home));
    const seed = gameSeed(L, g.home);
    if (!simUser && (g.home === L.userClub || g.away === L.userClub)) L.live = newGame(away, home, L.year, seed);
    else record(L, simulateGame(away, home, L.year, seed).state);
  }
  if (!L.live) L.day++;
}

export function closeLive(L: LeagueState): void {
  if (!L.live?.final) throw new Error('the live game is not over');
  record(L, L.live);
  L.live = null;
  L.day++;
}

export function simSeason(L: LeagueState): void {
  while (L.day < DAYS) playDay(L, true);
}

export interface Row { id: string; w: number; l: number; pct: number; gb: number; rs: number; ra: number }
export function standings(L: LeagueState): Row[] {
  const rows: Record<string, Row> = {};
  for (const id of Object.keys(L.clubs)) rows[id] = { id, w: 0, l: 0, pct: 0, gb: 0, rs: 0, ra: 0 };
  for (const r of L.results) {
    const h = rows[r.home]!, a = rows[r.away]!;
    h.rs += r.homeRuns; h.ra += r.awayRuns; a.rs += r.awayRuns; a.ra += r.homeRuns;
    if (r.homeRuns > r.awayRuns) { h.w++; a.l++; } else { a.w++; h.l++; }
  }
  const table = Object.values(rows).map((x) => ({ ...x, pct: x.w + x.l ? x.w / (x.w + x.l) : 0 }));
  table.sort((x, y) => y.pct - x.pct || (y.rs - y.ra) - (x.rs - x.ra) || x.id.localeCompare(y.id));
  const top = table[0]!;
  for (const x of table) x.gb = ((top.w - x.w) + (x.l - top.l)) / 2;
  return table;
}

export interface Leader { playerId: string; value: number }
export function leaders(L: LeagueState, pick: (s: PlayerSeason) => number, min: (s: PlayerSeason) => boolean, n = 5): Leader[] {
  return Object.entries(L.stats).filter(([, s]) => min(s)).map(([playerId, s]) => ({ playerId, value: pick(s) }))
    .sort((a, b) => b.value - a.value).slice(0, n);
}

export function clubOf(L: LeagueState, playerId: string): string | null {
  for (const c of Object.values(L.clubs)) if (c.roster.some((p) => p.id === playerId)) return c.id;
  return null;
}

import { city } from './cities';
import { autoLineup, autoRotation, hitters, pitchers } from './lineup';
import type { Club, Deal, LeaguePlayer, LeagueState } from './types';
import { askPrice, bidPrice, GREED, overall, SPREAD, value } from './valuation';

export const ROSTER_MAX = 25;
export type Verdict = { ok: true } | { ok: false; reason: string };
const no = (reason: string): Verdict => ({ ok: false, reason });
const UNTOUCHABLE = 'The Coalition promised the neighborhood he stays.';

const find = (club: Club, id: string): LeaguePlayer => {
  const p = club.roster.find((x) => x.id === id);
  if (!p) throw new Error(`${id} is not on ${club.id}`);
  return p;
};
const byOverall = (ps: LeaguePlayer[]) => [...ps].sort((a, b) => overall(b) - overall(a));

// Null when a roster can take the field; otherwise what is wrong.
export function rosterProblem(roster: LeaguePlayer[]): string | null {
  if (roster.length > ROSTER_MAX) return `more than ${ROSTER_MAX} players`;
  if (pitchers(roster).length < 5) return 'fewer than five pitchers';
  if (hitters(roster).length < 8) return 'fewer than eight position players';
  return null;
}

function move(from: Club, to: Club, ids: string[]): void {
  for (const id of ids) {
    const p = find(from, id);
    from.roster = from.roster.filter((x) => x !== p);
    to.roster.push(p);
  }
}
const after = (club: Club, out: string[], incoming: LeaguePlayer[]) => [...club.roster.filter((p) => !out.includes(p.id)), ...incoming];
function log(L: LeagueState, ai: Club, kind: Deal['kind'], given: number, received: number): void {
  ai.ledger.push({ year: L.year, partner: L.userClub, kind, given, received });
  autoRotation(ai);
  autoLineup(ai);
}
// An AI club over the cap drops its worst man into the pool (it never sits with 26).
function trim(L: LeagueState, club: Club): void {
  while (club.roster.length > ROSTER_MAX) {
    const cut = byOverall(club.roster).pop()!;
    club.roster = club.roster.filter((p) => p !== cut);
    L.pool.push({ ...cut, ask: value(cut) });
  }
}

// The user offers `give` (their ids) for `get` (the AI club's ids). The AI says yes when what it receives
// covers what it gives up plus its share of the spread.
export function proposeTrade(L: LeagueState, clubId: string, give: string[], get: string[]): Verdict {
  if (clubId === L.userClub) return no('you cannot trade with yourself');
  if (!give.length || !get.length) return no('a trade needs players on both sides');
  const me = L.clubs[L.userClub]!, them = L.clubs[clubId]!;
  const mine = give.map((id) => find(me, id)), theirs = get.map((id) => find(them, id));
  if (mine.some((p) => p.untouchable)) return no(UNTOUCHABLE);
  const p1 = rosterProblem(after(me, give, theirs));
  if (p1) return no(`your roster would have ${p1}`);
  const p2 = rosterProblem(after(them, get, mine));
  if (p2) return no(`${city(clubId).club} would have ${p2}`);
  const given = theirs.reduce((a, p) => a + value(p), 0), received = mine.reduce((a, p) => a + value(p), 0);
  if (received < given * (1 + SPREAD * GREED[city(clubId).personality])) return no(`${city(clubId).club} want more than that`);
  move(me, them, give);
  move(them, me, get);
  log(L, them, 'trade', given, received);
  return { ok: true };
}

export const salePrice = (L: LeagueState, _clubId: string, id: string) => bidPrice(value(find(L.clubs[L.userClub]!, id)));
export function sellPlayer(L: LeagueState, clubId: string, id: string): Verdict {
  if (clubId === L.userClub) return no('you cannot sell to yourself');
  const me = L.clubs[L.userClub]!, them = L.clubs[clubId]!;
  const p = find(me, id);
  if (p.untouchable) return no(UNTOUCHABLE);
  const price = salePrice(L, clubId, id);
  if (them.cash < price) return no(`${city(clubId).club} cannot afford $${price}`);
  if (city(clubId).personality === 'cheapskate' && price > them.cash / 4) return no(`${city(clubId).club} will not spend that much on one man`);
  const problem = rosterProblem(me.roster.filter((x) => x !== p));
  if (problem) return no(`your roster would have ${problem}`);
  move(me, them, [id]);
  me.cash += price;
  them.cash -= price;
  trim(L, them);
  log(L, them, 'sale', price, value(p));
  return { ok: true };
}

// Loyalists charge more; nobody sells a top-three man (a loyalist, top five).
export function purchasePrice(L: LeagueState, clubId: string, id: string): number {
  const base = askPrice(value(find(L.clubs[clubId]!, id)));
  return city(clubId).personality === 'loyalist' ? Math.round(base * 1.25) : base;
}
export function buyPlayer(L: LeagueState, clubId: string, id: string): Verdict {
  if (clubId === L.userClub) return no('you cannot buy from yourself');
  const me = L.clubs[L.userClub]!, them = L.clubs[clubId]!;
  const p = find(them, id);
  const keep = city(clubId).personality === 'loyalist' ? 5 : 3;
  if (byOverall(them.roster).slice(0, keep).includes(p)) return no(`${p.name} is not for sale`);
  const price = purchasePrice(L, clubId, id);
  if (me.cash < price) return no(`you cannot afford $${price}`);
  const problem = rosterProblem([...me.roster, p]) ?? rosterProblem(them.roster.filter((x) => x !== p));
  if (problem) return no(`a roster would have ${problem}`);
  move(them, me, [id]);
  me.cash -= price;
  them.cash += price;
  log(L, them, 'purchase', value(p), price);
  return { ok: true };
}

export function buyFromPool(L: LeagueState, id: string): Verdict {
  const me = L.clubs[L.userClub]!;
  const p = L.pool.find((x) => x.id === id);
  if (!p) return no('he has signed elsewhere');
  if (me.cash < p.ask) return no(`you cannot afford $${p.ask}`);
  if (me.roster.length >= ROSTER_MAX) return no(`you already carry ${ROSTER_MAX}`);
  L.pool = L.pool.filter((x) => x !== p);
  const { ask: _ask, ...player } = p;
  me.roster.push(player);
  me.cash -= p.ask;
  return { ok: true };
}

export function release(L: LeagueState, id: string): Verdict {
  const me = L.clubs[L.userClub]!;
  const p = find(me, id);
  if (p.untouchable) return no(UNTOUCHABLE);
  const problem = rosterProblem(me.roster.filter((x) => x !== p));
  if (problem) return no(`your roster would have ${problem}`);
  me.roster = me.roster.filter((x) => x !== p);
  L.pool.push({ ...p, ask: value(p) });
  return { ok: true };
}

import { next, type RngState } from '../core/rng';
import type { LeaguePlayer } from './types';
import { overall } from './valuation';

const clamp = (x: number) => Math.min(99, Math.max(20, Math.round(x)));
const step = (r: RngState, lo: number, hi: number) => lo + Math.floor(next(r) * (hi - lo + 1));
const BAT_KEYS = ['contact', 'power', 'eye', 'avoidK', 'fielding', 'arm'] as const;
const PIT_KEYS = ['control', 'stuff', 'contactSuppression', 'stamina'] as const;

// One offseason: +1 to +3 through 27 while under potential, flat at 28–29, −1 to −3 from 30.
// Speed declines from 28. Pitchers' batting never changes. Potential rises to meet an overshoot.
export function develop(p: LeaguePlayer, r: RngState): void {
  p.age++;
  const room = p.potential - overall(p);
  const grow = p.age <= 27 ? (room > 0 ? step(r, 1, 3) : 0) : p.age <= 29 ? 0 : -step(r, 1, 3);
  const skills = (p.pitching ?? p.batting) as unknown as Record<string, number>;
  for (const k of p.pitching ? PIT_KEYS : BAT_KEYS) skills[k] = clamp(skills[k]! + grow + step(r, -1, 1));
  if (!p.pitching && p.age >= 28) p.batting.speed = clamp(p.batting.speed - step(r, 1, 3));
  if (overall(p) > p.potential) p.potential = Math.ceil(overall(p));
}

export const retire = (p: LeaguePlayer) => p.age >= 38 || overall(p) < 40;

import type { LeagueState } from '../league/types';

export const SAVE_VERSION = 2;
const KEY = 'rafters.save';
const PREV = 'rafters.save.prev';

export interface KV { getItem(k: string): string | null; setItem(k: string, v: string): void }

export function serialize(L: LeagueState): string {
  return JSON.stringify(L);
}

// Version 1 was the M1 single-game save; it is refused on purpose (nothing worth migrating).
export function deserialize(json: string): LeagueState {
  const s = JSON.parse(json) as Partial<LeagueState> | null;
  if (!s || s.version !== SAVE_VERSION) throw new Error(`unsupported save version ${String(s?.version)}`);
  if (!s.clubs || !Array.isArray(s.schedule) || !s.rng || typeof s.day !== 'number') throw new Error('save is missing league state');
  return s as LeagueState;
}

export function writeSave(kv: KV, L: LeagueState): void {
  const current = kv.getItem(KEY);
  if (current !== null) {
    try { deserialize(current); kv.setItem(PREV, current); } catch { /* never rotate a bad save over a good one */ }
  }
  kv.setItem(KEY, serialize(L));
}

export function readSave(kv: KV): LeagueState | null {
  for (const key of [KEY, PREV]) {
    const raw = kv.getItem(key);
    if (raw === null) continue;
    try { return deserialize(raw); } catch { /* try the previous save */ }
  }
  return null;
}
